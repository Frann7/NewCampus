¿Cuántos renglones imprime este programa? 

#include <iostream>
#include <list>
#include <string>

class Suscriptor {
public:
    virtual ~Suscriptor() {}
    virtual void update(const std::string& mensaje) = 0;
};

class Canal {
private:
    std::list<Suscriptor*> suscriptores_;

public:
    void attach(Suscriptor* s) { suscriptores_.push_back(s); }
    void detach(Suscriptor* s) { suscriptores_.remove(s); }
    void notify(const std::string& mensaje) {
        for (Suscriptor* s : suscriptores_) {
            s->update(mensaje);
        }
    }
};

class Usuario : public Suscriptor {
private:
    std::string nombre_;

public:
    Usuario(const std::string& nombre) : nombre_(nombre) {}
    void update(const std::string& mensaje) override {
        std::cout << nombre_ << " recibe: " << mensaje << "\n";
    }
};

int main() {
    Canal canal;
    Usuario ana("Ana"), luis("Luis"), mia("Mia");
    canal.attach(&ana);
    canal.attach(&luis);
    canal.notify("video 1");
    canal.attach(&mia);
    canal.detach(&ana);
    canal.notify("video 2");
    return 0;
}

Inventada · Unidad 4C, Observer

Con el primer notify hay dos suscriptores (Ana y Luis): 2 renglones. Después se suma Mia y se da de baja Ana (detach), así que el segundo notify llega a Luis y a Mia: 2 renglones más. Total: 4. Ana no recibe "video 2": un suscriptor se puede sumar o sacar mientras el programa corre, sin tocar la clase Canal.
