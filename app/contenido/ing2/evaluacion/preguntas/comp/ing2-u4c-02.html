¿Qué patrón de diseño aplica este código?

#include <iostream>
#include <vector>

class Interesado {
public:
    virtual ~Interesado() {}
    virtual void actualizar(int precio) = 0;
};

class Accion {
private:
    std::vector<Interesado*> interesados_;
    int precio_ = 0;

public:
    void agregar(Interesado* interesado) { interesados_.push_back(interesado); }
    void fijarPrecio(int precio) {
        precio_ = precio;
        for (Interesado* i : interesados_) {
            i->actualizar(precio_);
        }
    }
};

class Pantalla : public Interesado {
public:
    void actualizar(int precio) override { std::cout << "Pantalla: " << precio << "\n"; }
};

class Alarma : public Interesado {
public:
    void actualizar(int precio) override {
        if (precio < 100) {
            std::cout << "Alarma: precio bajo\n";
        }
    }
};

int main() {
    Accion accion;
    Pantalla pantalla;
    Alarma alarma;
    accion.agregar(&pantalla);
    accion.agregar(&alarma);
    accion.fijarPrecio(120);
    accion.fijarPrecio(90);
    return 0;
}

Inventada · estilo Parcial 2025, Práctica 2

  • Composite
  • Observer
  • Strategy
  • Adapter

Accion guarda una lista de punteros a una interfaz (Interesado), tiene un método para agregar a esa lista y, cuando su precio cambia, recorre la lista llamando al mismo método de cada uno (actualizar()). Es Observer: Accion es el notificador (Subject) y Pantalla y Alarma son suscriptores. No es Composite: Accion no es un Interesado (la lista es de otro tipo) y del bucle no vuelve ningún resultado.
