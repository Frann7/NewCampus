¿Qué patrón de diseño aplica este código?

#include <iostream>
#include <string>

class Boton {
public:
    virtual ~Boton() {}
    virtual std::string dibujar() const = 0;
};
class Casilla {
public:
    virtual ~Casilla() {}
    virtual std::string dibujar() const = 0;
};

class BotonClaro : public Boton {
public:
    std::string dibujar() const override { return "boton claro"; }
};
class CasillaClara : public Casilla {
public:
    std::string dibujar() const override { return "casilla clara"; }
};
class BotonOscuro : public Boton {
public:
    std::string dibujar() const override { return "boton oscuro"; }
};
class CasillaOscura : public Casilla {
public:
    std::string dibujar() const override { return "casilla oscura"; }
};

class FabricaTema {
public:
    virtual ~FabricaTema() {}
    virtual Boton* crearBoton() const = 0;
    virtual Casilla* crearCasilla() const = 0;
};
class FabricaClara : public FabricaTema {
public:
    Boton* crearBoton() const override { return new BotonClaro(); }
    Casilla* crearCasilla() const override { return new CasillaClara(); }
};
class FabricaOscura : public FabricaTema {
public:
    Boton* crearBoton() const override { return new BotonOscuro(); }
    Casilla* crearCasilla() const override { return new CasillaOscura(); }
};

void armarPantalla(const FabricaTema& fabrica) {
    Boton* b = fabrica.crearBoton();
    Casilla* c = fabrica.crearCasilla();
    std::cout << b->dibujar() << " + " << c->dibujar() << "\n";
    delete b;
    delete c;
}

int main() {
    FabricaOscura oscura;
    armarPantalla(oscura);
    return 0;
}

Inventada · estilo Parcial 2025, Práctica 2

  • Factory Method
  • Abstract Factory
  • Strategy
  • Composite

FabricaTema es una interfaz con varios métodos de creación, uno por cada tipo de producto (crearBoton() y crearCasilla()), y cada fábrica concreta devuelve los productos de una misma familia (todos claros o todos oscuros). armarPantalla() recibe la fábrica y no nombra ninguna clase concreta. Es Abstract Factory. La diferencia con Factory Method: allá hay un método fábrica y un producto; acá, una fábrica entera para una familia.
