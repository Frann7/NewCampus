¿Qué imprime este programa?

#include <iostream>

class Contador {
private:
    static Contador* instancia_;
    int valor_;
    Contador() : valor_(0) {}

public:
    static Contador* getInstance() {
        if (instancia_ == nullptr) {
            instancia_ = new Contador();
        }
        return instancia_;
    }
    void sumar() { valor_++; }
    int valor() const { return valor_; }
};

Contador* Contador::instancia_ = nullptr;

int main() {
    Contador* a = Contador::getInstance();
    Contador* b = Contador::getInstance();
    a->sumar();
    b->sumar();
    b->sumar();
    std::cout << a->valor() << " " << (a == b) << "\n";
    return 0;
}

a y b apuntan al mismo objeto: la segunda llamada a getInstance() no crea nada, devuelve la instancia que ya existía. Por eso las tres sumas caen sobre el mismo contador (vale 3) y a == b da verdadero, que std::cout muestra como 1.
