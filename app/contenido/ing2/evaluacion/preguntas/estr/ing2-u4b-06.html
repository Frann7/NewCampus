¿Qué número imprime este programa? 

#include <iostream>
#include <vector>

class Elemento {
public:
    virtual ~Elemento() {}
    virtual int precio() const = 0;
};

class Producto : public Elemento {
private:
    int precio_;

public:
    Producto(int precio) : precio_(precio) {}
    int precio() const override { return precio_; }
};

class Caja : public Elemento {
private:
    std::vector<Elemento*> contenido_;

public:
    void agregar(Elemento* e) { contenido_.push_back(e); }
    int precio() const override {
        int total = 0;
        for (Elemento* e : contenido_) {
            total += e->precio();
        }
        return total;
    }
};

int main() {
    Producto a(100), b(50), c(25);
    Caja chica;
    chica.agregar(&c);
    Caja grande;
    grande.agregar(&a);
    grande.agregar(&b);
    grande.agregar(&chica);
    std::cout << grande.precio() << "\n";
    return 0;
}

Inventada · Unidad 4B, Composite

Se arma un árbol: la caja grande contiene el producto de 100, el de 50 y la caja chica, que a su vez contiene el de 25. grande.precio() le pide el precio a cada elemento de su lista; cuando le toca la caja chica, esta hace lo mismo con los suyos y devuelve 25. Total: 100 + 50 + 25 = 175. Esa llamada que baja por el árbol es lo que hace el Composite.
