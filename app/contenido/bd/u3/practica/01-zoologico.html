Guía 9 · CREATE TABLE

Todas las tablas del zoológico

Crear con SQL las tablas del modelo relacional del zoológico, con sus claves primarias y foráneas.

Empleado (Legajo (PK), Nombre, Direccion, Telefono, FechaIngreso)
Guia (Legajo (PK)(FK))
Cuidador (Legajo (PK)(FK))
Zona (Codigo (PK), Nombre, Extension)
Especie (Codigo (PK), Nombre, NombreCientifico, Descripcion, CodigoZona (FK))
Itinerario (Codigo (PK), Duracion, Longitud, CantVisitantes)
Cuidador_Especie (Legajo (PK)(FK), CodigoEspecie (PK)(FK), FechaDesde)
Guia_Itinerario (Legajo (PK)(FK), CodigoItinerario (PK)(FK), Hora)
Itinerario_Zona (CodigoItinerario (PK)(FK), CodigoZona (PK)(FK))
Paso 1 — El orden

Primero las que no tienen FK (Empleado, Zona, Itinerario), después las que apuntan a esas (Guia, Cuidador, Especie) y al final las intermedias.

Paso 2 — Los tipos
AtributoTipoPor qué
Legajo, Codigo, CantVisitantesINTNúmeros enteros
DuracionINTEn minutos, un entero
Extension, LongitudDECIMALMetros cuadrados y kilómetros, con decimales
TelefonoVARCHAR(20)No se hacen cuentas, puede tener guiones
FechaIngreso, FechaDesdeDATEFechas sin hora
HoraTIMESolo la hora del recorrido
Paso 3 — El script
Resolución 
CREATE TABLE Empleado (
  Legajo        INT           NOT NULL,
  Nombre        VARCHAR(100)  NOT NULL,
  Direccion     VARCHAR(150),
  Telefono      VARCHAR(20),
  FechaIngreso  DATE          NOT NULL,
  CONSTRAINT PK_Empleado PRIMARY KEY (Legajo)
);

CREATE TABLE Zona (
  Codigo     INT            NOT NULL,
  Nombre     VARCHAR(50)    NOT NULL,
  Extension  DECIMAL(10,2),
  CONSTRAINT PK_Zona PRIMARY KEY (Codigo)
);

CREATE TABLE Itinerario (
  Codigo          INT           NOT NULL,
  Duracion        INT,
  Longitud        DECIMAL(6,2),
  CantVisitantes  INT,
  CONSTRAINT PK_Itinerario PRIMARY KEY (Codigo)
);

CREATE TABLE Guia (
  Legajo  INT  NOT NULL,
  CONSTRAINT PK_Guia PRIMARY KEY (Legajo),
  CONSTRAINT FK_Guia_Empleado FOREIGN KEY (Legajo) REFERENCES Empleado(Legajo)
);

CREATE TABLE Cuidador (
  Legajo  INT  NOT NULL,
  CONSTRAINT PK_Cuidador PRIMARY KEY (Legajo),
  CONSTRAINT FK_Cuidador_Empleado FOREIGN KEY (Legajo) REFERENCES Empleado(Legajo)
);

CREATE TABLE Especie (
  Codigo            INT           NOT NULL,
  Nombre            VARCHAR(50)   NOT NULL,
  NombreCientifico  VARCHAR(100),
  Descripcion       VARCHAR(250),
  CodigoZona        INT           NOT NULL,
  CONSTRAINT PK_Especie PRIMARY KEY (Codigo),
  CONSTRAINT FK_Especie_Zona FOREIGN KEY (CodigoZona) REFERENCES Zona(Codigo)
);

CREATE TABLE Cuidador_Especie (
  Legajo         INT   NOT NULL,
  CodigoEspecie  INT   NOT NULL,
  FechaDesde     DATE  NOT NULL,
  CONSTRAINT PK_Cuidador_Especie PRIMARY KEY (Legajo, CodigoEspecie),
  CONSTRAINT FK_Cuidador_Especie_Cuidador FOREIGN KEY (Legajo) REFERENCES Cuidador(Legajo),
  CONSTRAINT FK_Cuidador_Especie_Especie FOREIGN KEY (CodigoEspecie) REFERENCES Especie(Codigo)
);

CREATE TABLE Guia_Itinerario (
  Legajo            INT   NOT NULL,
  CodigoItinerario  INT   NOT NULL,
  Hora              TIME  NOT NULL,
  CONSTRAINT PK_Guia_Itinerario PRIMARY KEY (Legajo, CodigoItinerario),
  CONSTRAINT FK_Guia_Itinerario_Guia FOREIGN KEY (Legajo) REFERENCES Guia(Legajo),
  CONSTRAINT FK_Guia_Itinerario_Itinerario FOREIGN KEY (CodigoItinerario) REFERENCES Itinerario(Codigo)
);

CREATE TABLE Itinerario_Zona (
  CodigoItinerario  INT  NOT NULL,
  CodigoZona        INT  NOT NULL,
  CONSTRAINT PK_Itinerario_Zona PRIMARY KEY (CodigoItinerario, CodigoZona),
  CONSTRAINT FK_Itinerario_Zona_Itinerario FOREIGN KEY (CodigoItinerario) REFERENCES Itinerario(Codigo),
  CONSTRAINT FK_Itinerario_Zona_Zona FOREIGN KEY (CodigoZona) REFERENCES Zona(Codigo)
);
Errores y trampas
  • En Guia y Cuidador, Legajo es PK y FK a la vez: lleva las dos líneas.
  • Guia_Itinerario.Legajo apunta a Guia, no a Empleado: así solo se pueden cargar recorridos de alguien que sea guía.
